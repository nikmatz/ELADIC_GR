Ο Lean είναι γλώσσα στην οποία γράφουμε μια απόδειξη τόσο αυστηρά, ώστε ο υπολογιστής ελέγχει κάθε βήμα. Αν ένα βήμα δεν προκύπτει από τα προηγούμενα, ο έλεγχος αποτυγχάνει.

1. Τι αποδεικνύεται: η διατύπωση στο βιβλίο και στον Lean

  • $(x^n)'=nx^{n-1}$
    theorem d_pow (x : ℝ) (n : ℕ) : HasDerivAt (fun y => y ^ n) (n * x ^ (n - 1)) x
  • $(e^x)'=e^x$
    theorem d_exp (x : ℝ) : HasDerivAt Real.exp (Real.exp x) x
  • $(a^x)'=a^x\ln a$, για $a\gt 0$
    theorem d_exp_base (a x : ℝ) (ha : 0 < a) :
    HasDerivAt (fun y => a ^ y) (a ^ x * Real.log a) x
  • $(\ln x)'=\dfrac{1}{x}$
    theorem d_log (x : ℝ) (hx : x ≠ 0) : HasDerivAt Real.log (1 / x) x
  • $(\log_a x)'=\dfrac{1}{x\ln a}$
    theorem d_logb (a x : ℝ) (hx : x ≠ 0) :
    HasDerivAt (fun y => Real.logb a y) (1 / (x * Real.log a)) x
  • $(\sqrt{x})'=\dfrac{1}{2\sqrt{x}}$
    theorem d_sqrt (x : ℝ) (hx : x ≠ 0) : HasDerivAt (fun y => √y) (1 / (2 * √x)) x
  • $(\sin x)'=\cos x$
    theorem d_sin (x : ℝ) : HasDerivAt Real.sin (Real.cos x) x
  • $(\cos x)'=-\sin x$
    theorem d_cos (x : ℝ) : HasDerivAt Real.cos (-Real.sin x) x
  • $(\tan x)'$${}=\sec^2x$${}=\dfrac{1}{\cos^2x}$
    theorem d_tan (x : ℝ) (hx : Real.cos x ≠ 0) : HasDerivAt Real.tan (1 / Real.cos x ^ 2) x
  • $(\cot x)'$${}=-\csc^2x$${}=-\dfrac{1}{\sin^2x}$
    theorem d_cot (x : ℝ) (hx : Real.sin x ≠ 0) :
    HasDerivAt Real.cot (-(1 / Real.sin x ^ 2)) x
  • $(\sec x)'$${}=\sec x\tan x$${}=\dfrac{\sin x}{\cos^2x}$
    theorem d_sec (x : ℝ) (hx : Real.cos x ≠ 0) :
    HasDerivAt (fun y => 1 / Real.cos y) (Real.sin x / Real.cos x ^ 2) x
  • $(\csc x)'$${}=-\csc x\cot x$${}=-\dfrac{\cos x}{\sin^2x}$
    theorem d_csc (x : ℝ) (hx : Real.sin x ≠ 0) :
    HasDerivAt (fun y => 1 / Real.sin y) (-Real.cos x / Real.sin x ^ 2) x
  • $(\arcsin x)'$${}=\dfrac{1}{\sqrt{1-x^2}}$
    theorem d_arcsin (x : ℝ) (h : |x| < 1) : HasDerivAt Real.arcsin (1 / √(1 - x ^ 2)) x
  • $(\arccos x)'$${}=-\dfrac{1}{\sqrt{1-x^2}}$
    theorem d_arccos (x : ℝ) (h : |x| < 1) : HasDerivAt Real.arccos (-(1 / √(1 - x ^ 2))) x
  • $(\arctan x)'=\dfrac{1}{1+x^2}$
    theorem d_arctan (x : ℝ) : HasDerivAt Real.arctan (1 / (1 + x ^ 2)) x
  • $(\sinh x)'=\cosh x$
    theorem d_sinh (x : ℝ) : HasDerivAt Real.sinh (Real.cosh x) x
  • $(\cosh x)'=\sinh x$
    theorem d_cosh (x : ℝ) : HasDerivAt Real.cosh (Real.sinh x) x

Το HasDerivAt f f' x σημαίνει ότι η $f$ έχει παράγωγο $f'$ στο $x$. Οι συναρτήσεις είναι αυτές της Mathlib (Real.exp, Real.log, Real.sin κ.λπ.)· η $\sec x$ γράφεται $\dfrac{1}{\cos x}$ και η $\csc x$ γράφεται $\dfrac{1}{\sin x}$. Κάθε τύπος έχει τις συνθήκες που χρειάζεται: $\cos x\neq 0$ για την $\tan$ και τη $\sec$, $\sin x\neq 0$ για την $\cot$ και τη $\csc$, $|x|\lt 1$ για τις $\arcsin$ και $\arccos$. Στον Lean ο Real.log ορίζεται και για $x\lt 0$, ως $\ln|x|$, και η διαίρεση με το $0$ δίνει $0$· γι’ αυτό οι τύποι για τον λογάριθμο και τη ρίζα διατυπώνονται για κάθε $x\neq 0$, και για $x\gt 0$ (με $a\gt 0$, $a\neq 1$) λένε ό,τι και το βιβλίο.

2. Πού βρίσκεται στη Mathlib

Χρησιμοποιήσαμε τα εξής θεωρήματα:

  • hasDerivAt_pow: η παράγωγος της $x^n$.
  • Real.hasDerivAt_exp: η παράγωγος της $e^x$.
  • Real.hasStrictDerivAt_const_rpow: η παράγωγος της $a^x$, για $a\gt 0$.
  • Real.hasDerivAt_log: η παράγωγος του $\ln x$.
  • Real.log_div_log: $\dfrac{\ln x}{\ln a}=\log_a x$.
  • Real.hasDerivAt_sqrt: η παράγωγος της $\sqrt{x}$.
  • Real.hasDerivAt_sin: η παράγωγος του $\sin x$.
  • Real.hasDerivAt_cos: η παράγωγος του $\cos x$.
  • Real.hasDerivAt_tan: η παράγωγος της $\tan x$.
  • Real.cot_eq_cos_div_sin: $\cot x=\dfrac{\cos x}{\sin x}$.
  • HasDerivAt.div: ο κανόνας του πηλίκου.
  • HasDerivAt.inv: η παράγωγος της $\dfrac{1}{f(x)}$.
  • Real.sin_sq_add_cos_sq: $\sin^2x+\cos^2x=1$.
  • Real.hasDerivAt_arcsin: η παράγωγος της $\arcsin x$.
  • Real.hasDerivAt_arccos: η παράγωγος της $\arccos x$.
  • Real.hasDerivAt_arctan: η παράγωγος της $\arctan x$.
  • Real.hasDerivAt_sinh: η παράγωγος του $\sinh x$.
  • Real.hasDerivAt_cosh: η παράγωγος του $\cosh x$.

import Mathlib

namespace Κεφ10.PinakasParagogon

theorem d_pow (x : ℝ) (n : ℕ) : HasDerivAt (fun y => y ^ n) (n * x ^ (n - 1)) x :=
  hasDerivAt_pow n x

theorem d_exp (x : ℝ) : HasDerivAt Real.exp (Real.exp x) x := Real.hasDerivAt_exp x

theorem d_exp_base (a x : ℝ) (ha : 0 < a) :
    HasDerivAt (fun y => a ^ y) (a ^ x * Real.log a) x :=
  (Real.hasStrictDerivAt_const_rpow ha x).hasDerivAt

theorem d_log (x : ℝ) (hx : x ≠ 0) : HasDerivAt Real.log (1 / x) x := by
  rw [one_div]
  exact Real.hasDerivAt_log hx

theorem d_logb (a x : ℝ) (hx : x ≠ 0) :
    HasDerivAt (fun y => Real.logb a y) (1 / (x * Real.log a)) x := by
  have h := (Real.hasDerivAt_log hx).div_const (Real.log a)
  have hfun : (fun y => Real.logb a y) = fun y => Real.log y / Real.log a := by
    funext y
    rw [Real.log_div_log]
  rw [hfun, one_div, mul_inv, ← div_eq_mul_inv]
  exact h

theorem d_sqrt (x : ℝ) (hx : x ≠ 0) : HasDerivAt (fun y => √y) (1 / (2 * √x)) x :=
  Real.hasDerivAt_sqrt hx

theorem d_sin (x : ℝ) : HasDerivAt Real.sin (Real.cos x) x := Real.hasDerivAt_sin x

theorem d_cos (x : ℝ) : HasDerivAt Real.cos (-Real.sin x) x := Real.hasDerivAt_cos x

theorem d_tan (x : ℝ) (hx : Real.cos x ≠ 0) : HasDerivAt Real.tan (1 / Real.cos x ^ 2) x :=
  Real.hasDerivAt_tan hx

theorem d_cot (x : ℝ) (hx : Real.sin x ≠ 0) :
    HasDerivAt Real.cot (-(1 / Real.sin x ^ 2)) x := by
  have h := (Real.hasDerivAt_cos x).div (Real.hasDerivAt_sin x) hx
  have hfun : Real.cos / Real.sin = Real.cot := by
    funext y
    rw [Pi.div_apply, Real.cot_eq_cos_div_sin]
  rw [hfun] at h
  convert h using 1
  rw [show -Real.sin x * Real.sin x - Real.cos x * Real.cos x = -1 by
    linear_combination -Real.sin_sq_add_cos_sq x]
  ring

theorem d_sec (x : ℝ) (hx : Real.cos x ≠ 0) :
    HasDerivAt (fun y => 1 / Real.cos y) (Real.sin x / Real.cos x ^ 2) x := by
  have h := (Real.hasDerivAt_cos x).inv hx
  rw [neg_neg] at h
  convert h using 1
  funext y
  rw [one_div, Pi.inv_apply]

theorem d_csc (x : ℝ) (hx : Real.sin x ≠ 0) :
    HasDerivAt (fun y => 1 / Real.sin y) (-Real.cos x / Real.sin x ^ 2) x := by
  have h := (Real.hasDerivAt_sin x).inv hx
  convert h using 1
  funext y
  rw [one_div, Pi.inv_apply]

theorem d_arcsin (x : ℝ) (h : |x| < 1) : HasDerivAt Real.arcsin (1 / √(1 - x ^ 2)) x := by
  rw [abs_lt] at h
  exact Real.hasDerivAt_arcsin h.1.ne' h.2.ne

theorem d_arccos (x : ℝ) (h : |x| < 1) : HasDerivAt Real.arccos (-(1 / √(1 - x ^ 2))) x := by
  rw [abs_lt] at h
  exact Real.hasDerivAt_arccos h.1.ne' h.2.ne

theorem d_arctan (x : ℝ) : HasDerivAt Real.arctan (1 / (1 + x ^ 2)) x :=
  Real.hasDerivAt_arctan x

theorem d_sinh (x : ℝ) : HasDerivAt Real.sinh (Real.cosh x) x := Real.hasDerivAt_sinh x

theorem d_cosh (x : ℝ) : HasDerivAt Real.cosh (Real.sinh x) x := Real.hasDerivAt_cosh x

end Κεφ10.PinakasParagogon
