Κριτήριο Λόγου (Ratio Test — d'Alembert)

Κεφάλαιο 19 · Εξήγηση από Μεγάλο Γλωσσικό Μοντέλο και απόδειξη ελεγμένη από υπολογιστή (Lean)

ΕκφώνησηΈστω σειρά $\sum a_n$ με $a_n\neq 0$ και έστω $L$${}=\lim_{n\to\infty}\left|\dfrac{a_{n+1}}{a_n}\right|$ (εφόσον το όριο υπάρχει). Αν $L<1$, η σειρά συγκλίνει απολύτως· αν $L>1$ (ή $L=+\infty$), αποκλίνει· αν $L=1$, το κριτήριο δεν αποφαίνεται. Το $L$${}=\lim\left|\dfrac{a_{n+1}}{a_n}\right|$ εμφανίζεται ως υπόθεση Tendsto (fun n => |a (n + 1) / a n|) atTop (𝓝 L), και το $L=+\infty$ ως Tendsto … atTop atTop. Η απόλυτη σύγκλιση γράφεται Summable a, γιατί για πραγματικούς αριθμούς αυτή η έννοια της Mathlib σημαίνει απόλυτη σύγκλιση· η απόκλιση γράφεται ως «τα μερικά αθροίσματα δεν έχουν όριο». Για τη σύγκλιση αρκεί $a_n\neq 0$ για μεγάλα $n$. Η συνθήκη $\limsup\left|\dfrac{a_{n+1}}{a_n}\right|\lt 1$ σημαίνει ότι υπάρχει $r\lt 1$ με $|a_{n+1}|\le r\,|a_n|$ για μεγάλα $n$, και έτσι διατυπώνεται. Για την απόκλιση η απόδειξη δείχνει ότι οι $|a_n|$ αυξάνουν από κάποιο $N$ και μετά, άρα δεν τείνουν στο $0$, και η αναγκαία συνθήκη αποτυγχάνει. Ότι για $L=1$ το κριτήριο δεν αποφαίνεται, και ότι από $\limsup\gt 1$ δεν έπεται απόκλιση, είναι παρατηρήσεις που φαίνονται με παραδείγματα και δεν διατυπώνονται εδώ.

2. Πού βρίσκεται στη Mathlib

Χρησιμοποιήσαμε τα εξής θεωρήματα:

  • summable_of_ratio_test_tendsto_lt_one: το κριτήριο λόγου για $L\lt 1$.
  • summable_of_ratio_norm_eventually_le: αν $|a_{n+1}|\le r\,|a_n|$ για μεγάλα $n$, με $r\lt 1$, η σειρά συγκλίνει απολύτως.
  • Nat.le_induction: επαγωγή για όλα τα $n\ge N$· δίνει $|a_n|\ge|a_N|$.
  • Filter.eventually_gt_atTop: για μεγάλα $x$ ισχύει $x\gt 1$· χρησιμοποιείται για το $L=+\infty$.

import Mathlib

namespace Κεφ19.KritirioLogou
open Filter Topology

-- L < 1: η σειρά συγκλίνει απολύτως
theorem ratio_converges (a : ℕ → ℝ) (L : ℝ) (hL : L < 1) (ha : ∀ᶠ n in atTop, a n ≠ 0)
    (h : Tendsto (fun n => |a (n + 1) / a n|) atTop (𝓝 L)) : Summable a := by
  refine summable_of_ratio_test_tendsto_lt_one hL ha ?_
  simpa [Real.norm_eq_abs, abs_div] using h

-- με limsup στη θέση του ορίου: |a (n + 1)| ≤ r |a n| για μεγάλα n, με r < 1
theorem ratio_converges_le (a : ℕ → ℝ) (r : ℝ) (hr : r < 1)
    (h : ∀ᶠ n in atTop, |a (n + 1)| ≤ r * |a n|) : Summable a :=
  summable_of_ratio_norm_eventually_le hr (by simpa [Real.norm_eq_abs] using h)

-- αν για μεγάλα n ο λόγος είναι > 1, τα μερικά αθροίσματα δεν συγκλίνουν
theorem diverges_of_ratio (a : ℕ → ℝ) (h : ∀ᶠ n in atTop, 1 < |a (n + 1) / a n|) :
    ¬ ∃ S, Tendsto (fun N => ∑ n ∈ Finset.range N, a n) atTop (𝓝 S) := by
  rintro ⟨S, hS⟩
  -- αναγκαία συνθήκη: a n → 0
  have h0 : Tendsto a atTop (𝓝 0) := by
    have h1 := (hS.comp (tendsto_add_atTop_nat 1)).sub hS
    simp only [Function.comp_def, Finset.sum_range_succ, add_sub_cancel_left, sub_self] at h1
    exact h1
  -- για n ≥ N ο λόγος είναι > 1
  obtain ⟨N, hN⟩ := eventually_atTop.mp h
  have hstep : ∀ n ≥ N, a n ≠ 0 ∧ |a n| < |a (n + 1)| := by
    intro n hn
    have h1 := hN n hn
    have ha : a n ≠ 0 := by
      intro h2
      rw [h2, div_zero, abs_zero] at h1
      linarith
    refine ⟨ha, ?_⟩
    rw [abs_div, lt_div_iff₀ (abs_pos.mpr ha), one_mul] at h1
    exact h1
  -- επομένως |a n| ≥ |a N| > 0 για κάθε n ≥ N
  have hge : ∀ n ≥ N, |a N| ≤ |a n| := by
    intro n hn
    induction n, hn using Nat.le_induction with
    | base => exact le_rfl
    | succ k hk ih => exact ih.trans (hstep k hk).2.le
  have hpos : 0 < |a N| := abs_pos.mpr (hstep N le_rfl).1
  -- αντίφαση με το a n → 0
  have h3 := h0.abs.eventually (gt_mem_nhds (show |(0 : ℝ)| < |a N| by
    rw [abs_zero]
    exact hpos))
  obtain ⟨n, hn1, hn2⟩ := (h3.and (eventually_ge_atTop N)).exists
  exact absurd (hge n hn2) (not_le.mpr hn1)

-- L > 1
theorem ratio_diverges (a : ℕ → ℝ) (L : ℝ) (hL : 1 < L)
    (h : Tendsto (fun n => |a (n + 1) / a n|) atTop (𝓝 L)) :
    ¬ ∃ S, Tendsto (fun N => ∑ n ∈ Finset.range N, a n) atTop (𝓝 S) :=
  diverges_of_ratio a (h.eventually (lt_mem_nhds hL))

-- L = +∞
theorem ratio_diverges_top (a : ℕ → ℝ) (h : Tendsto (fun n => |a (n + 1) / a n|) atTop atTop) :
    ¬ ∃ S, Tendsto (fun N => ∑ n ∈ Finset.range N, a n) atTop (𝓝 S) :=
  diverges_of_ratio a (h.eventually (eventually_gt_atTop 1))

end Κεφ19.KritirioLogou
