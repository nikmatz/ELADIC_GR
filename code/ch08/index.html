/* 1. Ορισμός και αποτίμηση */
define(f(x), x^3 - 3*x);
print("f(2)  =", f(2));
print("f(-1) =", f(-1));
print("f(0)  =", f(0));

/* 2. Μηδενικά (λύση f=0) */
zeros : solve(x^3 - 3*x = 0, x);
print("Μηδενικά f(x)=0:", zeros);

/* 3. Σύνθεση */
define(f1(x), x^2 + 1);
define(g1(x), 2*x - 3);
fog : f1(g1(x));
gof : g1(f1(x));
print("(fog)(x) =", expand(fog));
print("(gof)(x) =", expand(gof));

/* 4. Αντίστροφη f(x)=2x+5 */
inv_eq : solve(y = 2*x + 5, x);
print("f^-1(y) =", rhs(inv_eq[1]));

/* 5. Άρτια / Περιττή */
define(h1(x), x^4 - 2*x^2);
define(h2(x), x^3 - x);
print("h1(-x)-h1(x) =", expand(h1(-x)-h1(x)), " (0 => άρτια)");
print("h2(-x)+h2(x) =", expand(h2(-x)+h2(x)), " (0 => περιττή)");

/* 6. Γράφημα */
plot2d(x^3 - 3*x, [x,-3,3],
       [xlabel,"x"],[ylabel,"f(x)"],
       [title,"f(x) = x³ - 3x"]);
