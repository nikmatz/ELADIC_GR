📋 Περιεχόμενα Δραστηριοτήτων

Αντικείμενο: κρίσιμα σημεία και ακρότατα, πίνακας προσήμων, κυρτότητα και σημεία καμπής, Θεώρημα Μέσης Τιμής

Επιλέξτε γλώσσα από τις καρτέλες. Τον κώδικα Python μπορείτε να τον τρέξετε εδώ μέσα στη σελίδα (κουμπί «▶ Τρέξε») — μπορείτε και να τον επεξεργαστείτε. Ο κώδικας Maxima εκτελείται στο wxMaxima.

maxima_akrotata.mac ⬇ Λήψη .mac
Βασικές εντολές: solve(diff(f,x)=0,x)   subst(c   x   diff(f,x,2))   plot2d([f   diff(f,x)],[x,a,b])   float() 
/* ============================================================
   maxima_akrotata.mac
   Κεφάλαιο 11 — Μονοτονία, Ακρότατα, Κυρτότητα, Θ. Μέσης Τιμής
   Ματζάκος, Ν. (2026). Στοιχεία Γραμμικής Άλγεβρας, Διαφορικού & Ολοκληρωτικού Λογισμού. NewTech Publications.
   ============================================================

   ΕΝΤΟΛΕΣ ΠΟΥ ΧΡΗΣΙΜΟΠΟΙΟΥΝΤΑΙ:
     solve(diff(f,x)=0, x)        -> κρίσιμα σημεία
     subst(c, x, diff(f,x,2))     -> κριτήριο 2ης παραγώγου
     subst(t, x, diff(f,x))       -> πρόσημο της f' σε δοκιμαστική τιμή
     plot2d([f,diff(f,x)],[x,a,b])-> γραφική επαλήθευση
     float(...)                   -> δεκαδικές τιμές

   Εκτελέστε με: wxMaxima (File/Batch) ή
                 maxima -b maxima_akrotata.mac
   ============================================================ */

/* Η συνάρτηση του βιβλίου */
f  : x^3 - 3*x^2 - 9*x + 5$
f1 : diff(f, x)$
f2 : diff(f, x, 2)$

print("f(x)   =", f)$
print("f'(x)  =", f1, "  =", factor(f1))$
print("f''(x) =", f2, "  =", factor(f2))$

/* ── Α. Κρίσιμα σημεία και κριτήριο 2ης παραγώγου ───────── */
print("")$
print("════ Α. Κρίσιμα σημεία — κριτήριο 2ης παραγώγου ════")$

crit : sort(map(rhs, solve(f1 = 0, x)))$
print("  Κρίσιμα σημεία (f'=0): x =", crit, "   (αναμ. -1 και 3)")$

for c in crit do
  block([v : subst(c, x, f2), y : subst(c, x, f)],
    print("    x =", c, "  f(x) =", y, "  f''(x) =", v,
          if v > 0 then "  ->  τοπικό ΕΛΑΧΙΣΤΟ"
          else if v < 0 then "  ->  τοπικό ΜΕΓΙΣΤΟ"
          else "  ->  ασαφές (χρειάζεται άλλο κριτήριο)"))$

print("  Αναμενόμενα: f(-1)=10 τοπικό μέγιστο, f(3)=-22 τοπικό ελάχιστο.")$

/* ── Β. Μονοτονία: πίνακας προσήμων της f' με subst ─────── */
print("")$
print("════ Β. Μονοτονία — πίνακας προσήμων της f' ════")$

/* Τα κρίσιμα σημεία χωρίζουν το R σε τρία διαστήματα.
   Παίρνουμε μία δοκιμαστική τιμή μέσα σε κάθε διάστημα. */
tests : [-2, 0, 4]$                    /* (-inf,-1), (-1,3), (3,+inf) */
dlab : ["(-inf, -1)", "(-1, 3)", "(3, +inf)"]$

print("  Διάστημα          δοκιμ. x   f'(x)    πρόσημο   μονοτονία")$
for i:1 thru 3 do
  block([t : tests[i], v],
    v : subst(t, x, f1),
    print("  ", dlab[i], "     x =", t, "   f' =", v,
          if v > 0 then "   (+)   ΑΥΞΟΥΣΑ"
          else if v < 0 then "   (-)   ΦΘΙΝΟΥΣΑ"
          else "   (0)"))$

print("  Συμπέρασμα: αύξουσα στο (-inf,-1), φθίνουσα στο (-1,3),")$
print("              αύξουσα στο (3,+inf).")$
print("  => στο x=-1 αλλαγή + -> -  (μέγιστο), στο x=3 αλλαγή - -> + (ελάχιστο).")$

/* Γραφική επαλήθευση: f και f' μαζί */
plot2d([f, f1], [x, -4, 6],
       [legend, "f(x)", "f'(x)"],
       [title, "Κεφ.11: f(x)=x^3-3x^2-9x+5 και η f'(x)"])$

/* ── Γ. Σημεία καμπής και κυρτότητα / κοιλότητα ─────────── */
print("")$
print("════ Γ. Σημεία καμπής — κυρτότητα / κοιλότητα ════")$

infl : sort(map(rhs, solve(f2 = 0, x)))$
print("  Σημεία καμπής (f''=0): x =", infl, "   (αναμ. x = 1)")$
for c in infl do
  print("    x =", c, "  f(x) =", subst(c, x, f), "   (αναμ. f(1) = -6)")$

/* Πρόσημο της f'' εκατέρωθεν του σημείου καμπής */
test2 : [0, 2]$
dlab2 : ["(-inf, 1)", "(1, +inf)"]$
print("  Διάστημα          δοκιμ. x   f''(x)   κυρτότητα")$
for i:1 thru 2 do
  block([t : test2[i], v],
    v : subst(t, x, f2),
    print("  ", dlab2[i], "     x =", t, "   f'' =", v,
          if v > 0 then "    (+)   ΚΥΡΤΗ (convex)"
          else if v < 0 then "    (-)   ΚΟΙΛΗ (concave)"
          else "    (0)"))$

print("  Το πρόσημο της f'' αλλάζει στο x=1 => πράγματι σημείο καμπής.")$

/* Συγκεντρωτικό γράφημα f, f', f'' */
plot2d([f, f1, f2], [x, -4, 6],
       [legend, "f", "f'", "f''"],
       [title, "Κεφ.11: μελέτη της f(x)=x^3-3x^2-9x+5"])$

print("  Τιμές f στα κρίσιμα σημεία (δεκαδικά):",
      float(map(lambda([c], subst(c, x, f)), crit)))$

/* ── Δ. Θεώρημα Μέσης Τιμής: g(x)=x^2-x+1 στο [1,4] ─────── */
print("")$
print("════ Δ. Θεώρημα Μέσης Τιμής — g(x)=x^2-x+1 στο [1,4] ════")$

kill(c)$                               /* το c το χρειαζόμαστε ως μεταβλητή */
g  : x^2 - x + 1$
a  : 1$
b  : 4$
ga : subst(a, x, g)$
gb : subst(b, x, g)$
mslope : (gb - ga)/(b - a)$            /* κλίση χορδής */

print("  g(1) =", ga, "   g(4) =", gb)$
print("  Κλίση χορδής: (g(4)-g(1))/(4-1) =", mslope, "   (αναμ. 4)")$

/* Λύνουμε g'(c) = κλίση χορδής */
csol : map(rhs, solve(subst(c, x, diff(g, x)) = mslope, c))$
print("  Λύση της g'(c) = ", mslope, ":  c =", csol,
      " =", float(csol), "   (αναμ. c = 5/2)")$

cstar : first(csol)$
print("  Ανήκει στο ανοικτό (1,4);", is(cstar > a and cstar < b))$

gc : subst(cstar, x, g)$
chord   : expand(ga + mslope*(x - a))$              /* χορδή A(1,1)-B(4,13) */
tangent : expand(gc + mslope*(x - cstar))$          /* εφαπτομένη στο c     */

print("  g(c) =", gc, "   (αναμ. 19/4)")$
print("  Χορδή:      y =", chord,   "   (αναμ. 4x - 3)")$
print("  Εφαπτομένη: y =", tangent, "   (αναμ. 4x - 21/4)")$

/* Παράλληλες <=> ίδια κλίση <=> η διαφορά τους είναι σταθερή */
print("  Κλίση χορδής =", diff(chord, x),
      "   Κλίση εφαπτομένης =", diff(tangent, x))$
print("  Διαφορά (χορδή - εφαπτομένη) =", expand(chord - tangent),
      "  -> σταθερά => ΠΑΡΑΛΛΗΛΕΣ")$
print("  Έλεγχος d/dx (χορδή - εφαπτομένη) =", diff(chord - tangent, x),
      "  (πρέπει 0)")$

/* Γράφημα: g, χορδή, εφαπτομένη στο c  (άκρα σε αύξουσα σειρά) */
plot2d([g, chord, tangent], [x, 0.5, 4.5],
       [legend, "g(x)=x^2-x+1", "χορδή A(1,1)-B(4,13)",
                "εφαπτομένη στο c=5/2"],
       [title, "Θεώρημα Μέσης Τιμής: η εφαπτομένη στο c είναι παράλληλη στη χορδή"])$

print("")$
print("Ολοκληρώθηκε το Κεφάλαιο 11.")$
