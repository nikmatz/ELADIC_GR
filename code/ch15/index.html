/* ============================================================
   maxima_ttl.mac
   Κεφάλαιο 15 — Θεμελιώδες Θεώρημα Ολοκληρωτικού Λογισμού (ΘΘΟΛ)
                 Συνάρτηση εμβαδού και Κανόνας Leibniz
   Ματζάκος, Ν. (2026). Στοιχεία Γραμμικής Άλγεβρας, Διαφορικού & Ολοκληρωτικού Λογισμού. NewTech Publications.
   ============================================================

   ΕΝΤΟΛΕΣ ΠΟΥ ΧΡΗΣΙΜΟΠΟΙΟΥΝΤΑΙ:
     assume(x > 0)             → υπόθεση προσήμου (ΠΡΩΤΑ ΑΠ' ΟΛΑ!)
     integrate(f, t, 0, x)     → συνάρτηση εμβαδού F(x) = ∫₀ˣ f(t)dt
     diff(F, x)                → F'(x) — το Α΄ μέρος του ΘΘΟΛ
     subst(b, x, F)            → αντικατάσταση x := b (τύπος Newton–Leibniz)
     'integrate(f, t, a, b)    → ΑΝΕΚΤΕΛΕΣΤΟ ολοκλήρωμα· η diff εφαρμόζει
                                 πάνω του απευθείας τον κανόνα Leibniz
     solve(eq, c)              → επίλυση για το c του ΘΜΤ
     ratsimp, trigsimp, float  → απλοποίηση και δεκαδικές τιμές
     plot2d([...], [x, a, b])  → γραφική παράσταση

   ΠΡΟΣΟΧΗ: το assume(x > 0) μπαίνει ΠΡΙΝ από κάθε integrate με άνω
   όριο το x. Αλλιώς η Maxima ρωτά «Is x positive, negative or zero?»
   και σε εκτέλεση δέσμης (batch) μπλοκάρει για πάντα.

   Δραστηριότητα βιβλίου:
     (α) F(x)=∫₀ˣf(t)dt για f=t², sin t, 1/(1+t²)· έλεγχος F'(x)=f(x)
     (β) Newton–Leibniz: ∫₁³(2x+1), ∫₀^(π/2)cos x, ∫₁^e (ln x)/x
         πρώτα ως F(b)-F(a) με subst, μετά απευθείας
     (γ) κανόνας Leibniz με μεταβλητά άκρα
     (δ) ΘΜΤ Ολοκληρωτικού Λογισμού για f(x)=x² στο [0,3]

   Εκτελέστε με: wxMaxima (File → Batch File) ή
                 maxima -b maxima_ttl.mac
   ============================================================ */

/* ΠΡΩΤΑ ΑΠ' ΟΛΑ: οι υποθέσεις προσήμου — καμία ερώτηση στη συνέχεια */
assume(x > 0)$
print("Υποθέσεις σε ισχύ:", facts())$

/* ── Α. (α) Μέρος Α του ΘΘΟΛ: η συνάρτηση εμβαδού ───────── */
print("")$
print("════ Α. (α) Συνάρτηση εμβαδού F(x) = ∫₀ˣ f(t)dt ════")$

cases_a : [t^2, sin(t), 1/(1+t^2)]$

for ft in cases_a do
  block([F, dF, f_at_x],
    F      : integrate(ft, t, 0, x),
    dF     : trigsimp(ratsimp(diff(F, x))),
    f_at_x : subst(x, t, ft),
    print("  f(t) =", ft),
    print("     F(x) = ∫₀ˣ f(t)dt =", F),
    print("     F'(x) =", dF, "     F'(x) - f(x) =",
          ratsimp(trigsimp(dF - f_at_x)), "  (πρέπει 0)"))$

print("")$
print("  [Αναμένεται: f=t² → F=x³/3 ; f=sin t → F=1-cos x ;")$
print("               f=1/(1+t²) → F=atan(x)]")$

/* Τιμές της συνάρτησης εμβαδού (δείχνουν τη «συσσώρευση») */
print("")$
F_sin : integrate(sin(t), t, 0, x)$
print("Συσσώρευση για f(t)=sin t:")$
for a in [%pi/4, %pi/2, %pi, 3*%pi/2, 2*%pi] do
  print("   F(", a, ") =", ratsimp(subst(a, x, F_sin)),
        "=", float(subst(a, x, F_sin)))$
print("  [Μέγιστο στο x=π (εκεί f=sin x αλλάζει πρόσημο): F(π)=2]")$

/* ── Β. (β) Τύπος Newton–Leibniz: ∫_a^b f = F(b) - F(a) ─── */
print("")$
print("════ Β. (β) Newton–Leibniz — διπλό πέρασμα ════")$
print("Πρώτα ως F(b)-F(a) με subst, μετά απευθείας με integrate(f,x,a,b).")$

/* Δεδομένα του βιβλίου: [f(x), a, b, αναμενόμενη τιμή] */
nl_cases : [[2*x + 1,        1,     3,      10],
            [cos(x),         0,     %pi/2,  1],
            [log(x)/x,       1,     %e,     1/2]]$

for cs in nl_cases do
  block([fx, aa, bb, expect, G, val1, val2],
    fx     : cs[1],
    aa     : cs[2],
    bb     : cs[3],
    expect : cs[4],
    G      : integrate(fx, x),                       /* αντιπαράγωγος */
    val1   : ratsimp(subst(bb, x, G) - subst(aa, x, G)),   /* F(b)-F(a) */
    val2   : integrate(fx, x, aa, bb),                     /* απευθείας */
    print("  f(x) =", fx, "   στο [", aa, ",", bb, "]"),
    print("     Αντιπαράγωγος F(x) =", G),
    print("     (1) F(b) - F(a) =", val1, "=", float(val1)),
    print("     (2) integrate(f,x,a,b) =", val2, "=", float(val2)),
    print("     Ταυτίζονται; διαφορά =", ratsimp(val1 - val2),
          "   Αναμενόμενο:", expect,
          "   σωστό;", is(ratsimp(val1 - expect) = 0)))$

/* ── Γ. (γ) Κανόνας Leibniz με μεταβλητά άκρα ───────────── */
print("")$
print("════ Γ. (γ) Κανόνας Leibniz: F'(x) = f(b(x))b'(x) - f(a(x))a'(x) ════")$

/* 1. Ένα μεταβλητό άκρο: d/dx ∫₀^(x²) e^(-t²) dt                */
print("")$
print("(1)  d/dx ∫₀^(x²) e^(-t²) dt")$
D1   : ratsimp(diff('integrate(exp(-t^2), t, 0, x^2), x))$
rule1: subst(x^2, t, exp(-t^2))*diff(x^2, x) - subst(0, t, exp(-t^2))*diff(0, x)$
print("     Maxima (diff πάνω σε 'integrate):", D1)$
print("     Τύπος f(b(x))·b'(x) - f(a(x))·a'(x) :", ratsimp(rule1))$
print("     Διαφορά (πρέπει 0):", ratsimp(D1 - rule1))$
print("     [ΑΝΑΜΕΝΟΜΕΝΟ 2x·e^(-x⁴)]")$
/* Το ίδιο και με πραγματικό υπολογισμό του ολοκληρώματος (erf) */
print("     Με ρητό υπολογισμό του ολοκληρώματος:",
      ratsimp(diff(integrate(exp(-t^2), t, 0, x^2), x)))$

/* 2. ΔΥΟ μεταβλητά άκρα: d/dx ∫_x^(x²) sin(t²) dt              */
/*    Εδώ φαίνεται καθαρά ο ΑΡΝΗΤΙΚΟΣ όρος -f(a(x))·a'(x).      */
print("")$
print("(2)  d/dx ∫ₓ^(x²) sin(t²) dt   — ΔΥΟ μεταβλητά άκρα")$
D2   : ratsimp(diff('integrate(sin(t^2), t, x, x^2), x))$
rule2: subst(x^2, t, sin(t^2))*diff(x^2, x) - subst(x, t, sin(t^2))*diff(x, x)$
print("     Maxima (diff πάνω σε 'integrate):", D2)$
print("     Τύπος f(b(x))·b'(x) - f(a(x))·a'(x) :", ratsimp(rule2))$
print("     Διαφορά (πρέπει 0):", ratsimp(D2 - rule2))$
print("     [ΑΝΑΜΕΝΟΜΕΝΟ 2x·sin(x⁴) - sin(x²)]")$
print("     Ο όρος -sin(x²) προέρχεται από το ΚΑΤΩ άκρο a(x)=x")$
print("     (a'(x)=1) — χωρίς αυτόν ο τύπος θα ήταν λάθος.")$

/* Αριθμητικός έλεγχος με πεπερασμένες διαφορές στο x = 1.3 */
print("")$
print("Αριθμητικός έλεγχος στο x = 1.3 (κεντρική πεπερασμένη διαφορά):")$
block([x0, h, Iplus, Iminus, num, sym],
  x0 : 1.3,
  h  : 1.0e-4,
  Iplus  : quad_qags(sin(t^2), t, x0 + h, (x0 + h)^2)[1],
  Iminus : quad_qags(sin(t^2), t, x0 - h, (x0 - h)^2)[1],
  num : (Iplus - Iminus)/(2*h),
  sym : float(subst(x0, x, D2)),
  print("   πεπερασμένες διαφορές :", num),
  print("   συμβολικός τύπος      :", sym),
  print("   |διαφορά|             :", float(abs(num - sym))))$

/* ── Δ. (δ) ΘΜΤ Ολοκληρωτικού Λογισμού ──────────────────── */
print("")$
print("════ Δ. (δ) ΘΜΤ Ολοκληρωτικού Λογισμού: f(x)=x² στο [0,3] ════")$

kill(c)$
a_mvt : 0$
b_mvt : 3$
I_mvt : integrate(x^2, x, a_mvt, b_mvt)$
print("∫₀³ x² dx =", I_mvt, "   [ΑΝΑΜΕΝΟΜΕΝΟ 9]")$

print("Ζητάμε c ώστε f(c)·(b-a) = ∫ₐᵇ f  ⇔  c²·3 = 9:")$
sols : solve(c^2*(b_mvt - a_mvt) = I_mvt, c)$
print("   solve →", sols)$
c_val : rhs(second(sols))$
print("   Δεκτή λύση στο [0,3]: c =", c_val, "=", float(c_val),
      "   [ΑΝΑΜΕΝΟΜΕΝΟ √3 ≈ 1.7320508]")$
print("   f(c) =", subst(c_val, c, c^2), "   ύψος του ορθογωνίου")$
print("   Έλεγχος f(c)·(b-a) - ∫ₐᵇf =",
      ratsimp(subst(c_val, c, c^2)*(b_mvt - a_mvt) - I_mvt), "  (πρέπει 0)")$
print("   Εμβαδόν ορθογωνίου =", float(subst(c_val, c, c^2)*(b_mvt - a_mvt)),
      "   Εμβαδόν κάτω από την f =", float(I_mvt))$

/* Γραφικά: η f, η οριζόντια ευθεία ύψους f(c) = 3, και το c */
h_rect : subst(c_val, c, c^2)$
plot2d([x^2, h_rect], [x, 0, 3], [y, 0, 9.5],
       [legend, "f(x) = x²", "y = f(c) = 3 (ύψος ορθογωνίου)"],
       [xlabel, "x"], [ylabel, "y"],
       [title, "ΘΜΤ Ολοκληρωτικού Λογισμού: ισότητα εμβαδών, c = sqrt(3)"])$

print("")$
print("Τέλος — Κεφάλαιο 15.")$
