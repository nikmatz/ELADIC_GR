/* 1. Εμβαδόν μεταξύ δύο καμπυλών */
f: x^2$
g: x+2$
pts: solve(f = g, x);
[a, b]: [rhs(pts[1]), rhs(pts[2])];
A: integrate(g - f, x, a, b);
print("Εμβαδόν =", A);

/* Γράφημα */
plot2d([f, g, [parametric, t, 0, [t, a, b]]],
       [x, a-0.5, b+0.5],
       [style, [lines,2,1], [lines,2,2], [lines,1,3]],
       [legend, "f(x)=x²", "g(x)=x+2"],
       [title, "Εμβαδόν μεταξύ δύο καμπυλών"])$

/* 2. Όγκος περιστροφής (δίσκοι) */
f2: sqrt(x)$
V: %pi * integrate(f2^2, x, 0, 4);
print("Όγκος V =", V, "≈", float(V));

/* 3. Μήκος καμπύλης */
f3: x^(3/2)$
df3: diff(f3, x)$
L: integrate(sqrt(1 + df3^2), x, 0, 4);
print("Μήκος L =", ratsimp(L));

/* 4. Ροπή αδράνειας */
assume(rho > 0, L_len > 0)$
I: rho * integrate(x^2, x, 0, L_len);
print("Ροπή αδράνειας I =", I);
