/* ============================================================ */
/* maxima_efarm_ol.mac                                          */
/* Κεφάλαιο 17 — Εφαρμογές Ολοκληρωτικού Λογισμού               */
/*   Εμβαδόν, Όγκος Περιστροφής, Μήκος Τόξου                    */
/* Ματζάκος, Ν. (2026). Στοιχεία Γραμμικής Άλγεβρας, Διαφορικού & Ολοκληρωτικού Λογισμού. NewTech Publications. */
/* ------------------------------------------------------------ */
/* ΒΑΣΙΚΕΣ ΕΝΤΟΛΕΣ:                                             */
/*   integrate(f,x,a,b)  -> ορισμένο ολοκλήρωμα                 */
/*   romberg(f,x,a,b)    -> αριθμητικό ολοκλήρωμα               */
/*   solve(f=g,x)        -> σημεία τομής                        */
/*   diff(f,x), sqrt()   -> μήκος τόξου / επιφάνεια             */
/*   trigsimp(), ratsimp()                                      */
/* ΠΡΟΣΟΧΗ: τα σχόλια στη Maxima ανοίγουν με κάθετο-αστερίσκο     */
/*   και κλείνουν με αστερίσκο-κάθετο. Ποτέ με ερωτηματικά.     */
/* ============================================================ */

load(romberg)$
/* Σφίγγουμε την ανοχή της romberg (προεπιλογή 1e-4) ώστε τα     */
/* αριθμητικά αποτελέσματα να συμφωνούν με τα συμβολικά.         */
rombergtol : 1e-10$
rombergit  : 20$

print("============================================================")$
print(" Κεφάλαιο 17: Εφαρμογές Ολοκληρωτικού Λογισμού")$
print("============================================================")$

/* ============================================================ */
/* Α. ΕΜΒΑΔΟΝ                                                   */
/* ============================================================ */
print("")$
print("=== Α. ΕΜΒΑΔΟΝ ===")$

/* Α1.  int_{-1}^{1} |x^3 - x| dx                               */
/* Η Maxima δεν ολοκληρώνει απευθείας την απόλυτη τιμή:         */
/* βρίσκουμε τις ρίζες, τις ΤΑΞΙΝΟΜΟΥΜΕ και σπάμε το διάστημα.  */
print("")$
print("Α1. int_{-1}^{1} |x^3 - x| dx")$
rizes : sort(map(rhs, solve(x^3 - x = 0, x)))$
print("   ρίζες της x^3-x (ταξινομημένες):", rizes, " (αναμ. [-1,0,1])")$
/* Στο [-1,0] είναι x^3-x >= 0, στο [0,1] είναι x^3-x <= 0 */
A1a : integrate(x^3 - x, x, -1, 0)$
A1b : integrate(-(x^3 - x), x, 0, 1)$
print("   int_{-1}^{0} (x^3-x) dx  =", A1a)$
print("   int_{0}^{1} -(x^3-x) dx  =", A1b)$
A1 : A1a + A1b$
print("   ΣΥΝΟΛΟ (integrate)       =", A1, "=", float(A1), " (αναμ. 1/2)")$
A1r : romberg(abs(x^3 - x), x, -1, 1)$
print("   ΣΥΝΟΛΟ (romberg)         =", A1r)$
print("   integrate vs romberg, |διαφορά| < 1e-8 :",
      is(abs(float(A1) - A1r) < 1e-8))$

/* Α2.  Εμβαδόν μεταξύ y = -x^2+4 και y = x^2-2x                */
print("")$
print("Α2. Εμβαδόν μεταξύ y = -x^2+4 και y = x^2-2x")$
f : -x^2 + 4$
g : x^2 - 2*x$
/* ΠΑΝΤΑ ταξινομούμε τις ρίζες: η solve τις επιστρέφει [2,-1],  */
/* και με [a,b]=[2,-1] θα βγει ΑΡΝΗΤΙΚΟ εμβαδόν (και θα σκάσει  */
/* η plot2d, που απαιτεί min < max).                            */
tomes : sort(map(rhs, solve(f = g, x)))$
print("   solve(f=g,x) ταξινομημένα:", tomes, " (αναμ. [-1,2])")$
a : tomes[1]$
b : tomes[2]$
print("   a =", a, ",  b =", b)$
A2 : integrate(f - g, x, a, b)$
print("   Εμβαδόν =", A2, "=", float(A2), " (αναμ. 9)")$
print("   θετικό εμβαδόν:", is(A2 > 0))$
A2r : romberg(f - g, x, float(a), float(b))$
print("   Εμβαδόν (romberg) =", A2r)$
print("   integrate vs romberg, |διαφορά| < 1e-8 :",
      is(abs(float(A2) - A2r) < 1e-8))$

/* Γράφημα: τα άκρα ΠΑΝΤΑ σε αύξουσα σειρά */
plot2d([f, g],
       [x, a - 0.5, b + 0.5],
       [style, [lines, 2, 1], [lines, 2, 2]],
       [legend, "f(x) = -x^2+4", "g(x) = x^2-2x"],
       [xlabel, "x"], [ylabel, "y"],
       [title, "Εμβαδόν μεταξύ δύο καμπυλών (= 9)"])$

/* ============================================================ */
/* Β. ΟΓΚΟΣ ΠΕΡΙΣΤΡΟΦΗΣ                                         */
/* ============================================================ */
print("")$
print("=== Β. ΟΓΚΟΣ ΠΕΡΙΣΤΡΟΦΗΣ ===")$

/* Β1. ΔΙΣΚΟΙ:  V = pi * int_a^b [f(x)]^2 dx                    */
print("")$
print("Β1. Μέθοδος δίσκων:  V = pi*int_a^b f(x)^2 dx")$
fd1 : sqrt(x)$
Vd1 : %pi * integrate(fd1^2, x, 0, 4)$
print("   y = sqrt(x), [0,4] : V =", Vd1, "=", float(Vd1), " (αναμ. 8*pi)")$
print("   έλεγχος V = 8*pi   :", is(ratsimp(Vd1 - 8*%pi) = 0))$

fd2 : sin(x)$
Vd2 : %pi * integrate(fd2^2, x, 0, %pi)$
print("   y = sin(x), [0,pi] : V =", Vd2, "=", float(Vd2), " (αναμ. pi^2/2)")$
print("   έλεγχος V = pi^2/2 :", is(ratsimp(Vd2 - %pi^2/2) = 0))$

/* Β2. ΔΑΚΤΥΛΙΟΙ: V = pi*int_a^b ([R]^2 - [r]^2) dx             */
/*     y = sqrt(x) (εξωτερική) vs y = x (εσωτερική),            */
/*     περιστροφή γύρω από τον x-άξονα.                         */
print("")$
print("Β2. Μέθοδος δακτυλίων: y = sqrt(x) vs y = x, γύρω από τον x-άξονα")$
/* Η solve δεν χειρίζεται καλά τη ρίζα: υψώνουμε στο τετράγωνο  */
/* (sqrt(x) = x  <=>  x = x^2 για x >= 0) και ΤΑΞΙΝΟΜΟΥΜΕ.      */
tomes2 : sort(map(rhs, solve(x = x^2, x)))$
print("   τομές (ταξινομημένες):", tomes2, " (αναμ. [0,1])")$
R_ex : sqrt(x)$
r_es : x$
Vw : %pi * integrate(R_ex^2 - r_es^2, x, tomes2[1], tomes2[2])$
print("   V =", Vw, "=", float(Vw), " (αναμ. pi/6)")$
print("   έλεγχος V = pi/6 :", is(ratsimp(Vw - %pi/6) = 0))$
/* Διασταύρωση με κελύφη ως προς y: ακτίνα y, ύψος (y - y^2) */
Vw_shell : 2*%pi * integrate(y*(y - y^2), y, 0, 1)$
print("   ίδιο με κελύφη ως προς y:", Vw_shell,
      " ταυτίζονται:", is(ratsimp(Vw - Vw_shell) = 0))$

/* Β3. ΚΕΛΥΦΗ: V = 2*pi*int_a^b x*f(x) dx (γύρω από τον y-άξονα) */
print("")$
print("Β3. Μέθοδος κελυφών: y = x^3 γύρω από τον y-άξονα, [0,2]")$
fs : x^3$
Vs : 2*%pi * integrate(x*fs, x, 0, 2)$
print("   V = 2*pi*int_0^2 x*x^3 dx =", Vs, "=", float(Vs), " (αναμ. 64*pi/5)")$
print("   έλεγχος V = 64*pi/5 :", is(ratsimp(Vs - 64*%pi/5) = 0))$
/* Διασταύρωση με δακτυλίους ως προς y (y από 0 ως 8, x = y^(1/3)) */
/* Το y > 0 δηλώνεται ΠΡΙΝ, ώστε να μη ρωτήσει η Maxima.          */
assume(y > 0)$
Vs_wash : %pi * integrate(2^2 - (y^(1/3))^2, y, 0, 8)$
print("   ίδιο με δακτυλίους ως προς y:", ratsimp(Vs_wash),
      " ταυτίζονται:", is(ratsimp(Vs - Vs_wash) = 0))$
forget(y > 0)$

/* ============================================================ */
/* Γ. ΜΗΚΟΣ ΤΟΞΟΥ:  L = int_a^b sqrt(1 + [f'(x)]^2) dx          */
/* ============================================================ */
print("")$
print("=== Γ. ΜΗΚΟΣ ΤΟΞΟΥ ===")$

/* Γ1.  y = x^(3/2) στο [0,4] */
print("")$
print("Γ1. y = x^(3/2), [0,4]")$
y1 : x^(3/2)$
dy1 : diff(y1, x)$
print("   f'(x) =", dy1)$
ds1 : sqrt(1 + dy1^2)$
print("   sqrt(1+f'^2) =", ratsimp(ds1))$
L1 : ratsimp(integrate(ds1, x, 0, 4))$
print("   L =", L1, "=", float(L1), " (αναμ. (80*sqrt(10)-8)/27 = 9.073415289)")$
print("   έλεγχος κλειστής μορφής:", is(ratsimp(L1 - (80*sqrt(10)-8)/27) = 0))$
L1r : romberg(ds1, x, 0, 4)$
print("   L (romberg) =", L1r)$
print("   integrate vs romberg, |διαφορά| < 1e-8 :",
      is(abs(float(L1) - L1r) < 1e-8))$

/* Γ2.  y = ln(cos x) στο [0, pi/3] — εδώ χρειάζεται trigsimp   */
/*      1 + tan^2(x) = sec^2(x), άρα sqrt(1+f'^2) = sec(x).     */
/*      Δηλώνουμε cos(x)>0 ΠΡΙΝ, ώστε sqrt(1/cos^2) = 1/cos.    */
assume(cos(x) > 0)$
print("")$
print("Γ2. y = ln(cos x), [0, pi/3]")$
y2 : log(cos(x))$
dy2 : diff(y2, x)$
print("   f'(x) =", dy2, " (= -tan x)")$
ds2raw : 1 + dy2^2$
ds2 : trigsimp(ds2raw)$
print("   trigsimp(1 + f'^2) =", ds2, " (= sec^2 x)")$
L2 : integrate(sqrt(ds2), x, 0, %pi/3)$
print("   L =", L2)$
print("   L =", float(L2), " (αναμ. ln(2+sqrt(3)) = 1.316957897)")$
print("   έλεγχος L = ln(2+sqrt(3)) :",
      is(abs(float(L2) - float(log(2 + sqrt(3)))) < 1e-9))$
L2r : romberg(sqrt(ds2), x, 0, float(%pi/3))$
print("   L (romberg) =", L2r)$
print("   integrate vs romberg, |διαφορά| < 1e-8 :",
      is(abs(float(L2) - L2r) < 1e-8))$
forget(cos(x) > 0)$

/* ============================================================ */
/* Δ. ΕΜΒΑΔΟΝ ΕΠΙΦΑΝΕΙΑΣ ΕΚ ΠΕΡΙΣΤΡΟΦΗΣ                         */
/*    S = 2*pi * int_a^b f(x)*sqrt(1 + f'(x)^2) dx              */
/* ============================================================ */
print("")$
print("=== Δ. ΕΜΒΑΔΟΝ ΕΠΙΦΑΝΕΙΑΣ (κώνος y = x, [0,2]) ===")$
fc : x$
dfc : diff(fc, x)$
print("   f(x) =", fc, ",  f'(x) =", dfc)$
print("   sqrt(1+f'^2) =", sqrt(1 + dfc^2), " (= sqrt(2))")$
S : 2*%pi * integrate(fc*sqrt(1 + dfc^2), x, 0, 2)$
print("   S = 2*pi*int_0^2 x*sqrt(2) dx =", S, "=", float(S),
      " (αναμ. 4*sqrt(2)*pi = 17.77153175)")$
/* Επαλήθευση με τον γεωμετρικό τύπο S = pi*r*l */
r_kwnou : 2$
l_kwnou : 2*sqrt(2)$
S_typos : %pi * r_kwnou * l_kwnou$
print("   Τύπος S = pi*r*l, r =", r_kwnou, ", l =", l_kwnou, " -> S =",
      S_typos, "=", float(S_typos))$
print("   ταυτίζονται:", is(ratsimp(S - S_typos) = 0))$

/* ============================================================ */
/* Ε. ΦΥΣΙΚΕΣ ΕΦΑΡΜΟΓΕΣ                                         */
/* ============================================================ */
print("")$
print("=== Ε. ΦΥΣΙΚΕΣ ΕΦΑΡΜΟΓΕΣ ===")$

/* Ε1. Έργο ελατηρίου:  W = int_0^{0.1} F(x) dx,  F = 200x (N)  */
print("")$
print("Ε1. Έργο ελατηρίου, F(x) = 200*x N, x από 0 έως 0,1 m")$
F_elat : 200*x$
W : integrate(F_elat, x, 0, 1/10)$
print("   W = int_0^{1/10} 200*x dx =", W, "J =", float(W), "J  (αναμ. 1 J)")$
print("   έλεγχος W = 1 J:", is(W = 1))$
/* Ισοδύναμα από τον τύπο W = k*x^2/2 με k = 200, x = 0,1 */
print("   τύπος k*x^2/2 =", 200*(1/10)^2/2, "J  ταυτίζεται:",
      is(W - 200*(1/10)^2/2 = 0))$

/* Ε2. Κέντρο βάρους του χωρίου κάτω από την y = x^2, [0,3]     */
/*     A  = int_0^3 f dx                                        */
/*     xb = (1/A)*int_0^3 x*f dx                                */
/*     yb = (1/A)*int_0^3 (1/2)*f^2 dx                          */
print("")$
print("Ε2. Κέντρο βάρους του χωρίου κάτω από την y = x^2, [0,3]")$
fy : x^2$
Ar : integrate(fy, x, 0, 3)$
xb : integrate(x*fy, x, 0, 3) / Ar$
yb : integrate(fy^2/2, x, 0, 3) / Ar$
print("   Εμβαδόν A =", Ar, " (αναμ. 9)")$
print("   x_bar =", xb, "=", float(xb), " (αναμ. 9/4 = 2.25)")$
print("   y_bar =", yb, "=", float(yb), " (αναμ. 27/10 = 2.7)")$
print("   έλεγχοι:", is(Ar = 9), is(xb = 9/4), is(yb = 27/10))$

print("")$
print("Ολοκληρώθηκε.")$
