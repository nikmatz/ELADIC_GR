/* 1. Όρια ακολουθιών */
print("limit n/(n+1)   =", limit(n/(n+1), n, inf));
print("limit (1+1/n)^n =", limit((1+1/n)^n, n, inf));
print("limit n^2/exp(n)=", limit(n^2/exp(n), n, inf));
print("limit sin(n)/n  =", limit(sin(n)/n, n, inf));

/* 2. Γεωμετρική σειρά */
for q in [1/2, 1/3, 2/3] do (
  S: sum(q^k, k, 0, inf),
  print("q =", q, "->", S)
)$

/* 3. Taylor / Maclaurin */
taylor(exp(x),   x, 0, 7);
taylor(sin(x),   x, 0, 7);
taylor(cos(x),   x, 0, 7);
taylor(1/(1-x),  x, 0, 6);
taylor(log(1+x), x, 0, 6);

/* 4. Κριτήριο λόγου: n!/n^n */
a[n] := n! / n^n$
ratio: limit(a[n+1]/a[n], n, inf);
print("Κριτήριο λόγου L =", ratio, " -> σύγκλιση");

/* 5. Σειρά ισχύος — ακτίνα σύγκλισης */
c[n] := 1/n^2$
R: 1 / limit(c[n+1]/c[n], n, inf);
print("Ακτίνα R =", R);

/* 6. Γράφημα μερικών αθροισμάτων Taylor sin(x) */
approx1: x - x^3/6;
approx3: x - x^3/6 + x^5/120 - x^7/5040;
plot2d([sin(x), approx1, approx3], [x, -3.14, 3.14],
       [legend, "sin(x)", "Τάξη 3", "Τάξη 7"],
       [title, "Taylor sin(x)"])$

/* ============================================================ */
/* ΣΥΜΠΛΗΡΩΜΑ — Όροι, μερικά αθροίσματα και ο ε-N ορισμός       */
/*   makelist, abs, float                                        */
/* ============================================================ */

/* 1. Οι πρώτοι όροι μιας ακολουθίας με makelist                */
a(n) := (2*n + 1)/(n + 3)$
print("Πρώτοι 8 όροι:", float(makelist(a(n), n, 1, 8)))$
print("Όριο:", limit(a(n), n, inf))$

/* 2. Ο ε-N ορισμός αριθμητικά: πόσο μεγάλο πρέπει να είναι το n; */
L : limit(a(n), n, inf)$
print("")$
for eps in [0.1, 0.01, 0.001] do
  block([N : 1],
    while abs(float(a(N) - L)) >= eps do N : N + 1,
    print("  ε =", eps, " -> πρώτο N με |a_N - L| < ε :", N,
          "  |a_N-L| =", float(abs(a(N)-L))))$

/* 3. Μερικά αθροίσματα με makelist                              */
print("")$
S(N) := sum(1/2^k, k, 0, N)$
print("Γεωμετρική Σ(1/2)^k, S_0..S_8:", float(makelist(S(N), N, 0, 8)))$
print("Άθροισμα:", ev(sum(1/2^k, k, 0, inf), simpsum))$

/* 4. Αρμονική: αποκλίνει αργά                                   */
print("")$
H(N) := sum(1/k, k, 1, N)$
for N in [10, 100, 500] do
  print("  H_", N, "=", float(H(N)), "   log(N)+γ ~",
        float(log(N) + 0.5772156649))$

/* 5. Εναλλάσσουσα: το σφάλμα φράσσεται από τον πρώτο           */
/*    παραλειπόμενο όρο  |S - S_N| <= b_{N+1}                    */
print("")$
A(N) := sum((-1)^(k+1)/k, k, 1, N)$
for N in [10, 50, 100] do

        "   b_(N+1) =", float(1/(N+1)))$
