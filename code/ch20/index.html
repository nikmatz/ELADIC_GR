/* Εντολές: taylor(f,x,x0,n), diff(T,x), integrate(T,x), limit(), sum() με simpsum */

print("============================================================")$
print("Α. ΑΚΤΙΝΑ ΣΥΓΚΛΙΣΗΣ   R = lim |c_n / c_(n+1)|")$
print("============================================================")$

/* Υπολογίζουμε κατευθείαν το lim |c_n/c_(n+1)| (ο τύπος του βιβλίου):
   έτσι δεν χρειάζεται διαίρεση 1/L και δεν σκάει όταν L = 0.        */
Rconv(cn) := limit(abs(cn / subst(n+1, n, cn)), n, inf)$

print("Σ x^n/n   : R =", Rconv(1/n),   "   (αναμενόμενο 1)")$
print("Σ x^n/n!  : R =", Rconv(1/n!),  "   (αναμενόμενο inf)")$
print("Σ n x^n   : R =", Rconv(n),     "   (αναμενόμενο 1)")$
print("Σ x^n/2^n : R =", Rconv(1/2^n), "   (αναμενόμενο 2)")$
print("Σ n! x^n  : R =", Rconv(n!),    "   (αναμενόμενο 0)")$
print("")$
print("R = inf : συγκλίνει παντού.  R = 0 : μόνο στο x = 0.")$

print("")$
print("============================================================")$
print("Β. ΑΝΑΠΤΥΓΜΑΤΑ MACLAURIN  taylor(f, x, 0, 8)")$
print("============================================================")$

print("e^x       =", taylor(exp(x),    x, 0, 8))$
print("sin(x)    =", taylor(sin(x),    x, 0, 8))$
print("cos(x)    =", taylor(cos(x),    x, 0, 8))$
print("log(1+x)  =", taylor(log(1+x),  x, 0, 8))$
print("1/(1-x)   =", taylor(1/(1-x),   x, 0, 8))$
print("atan(x)   =", taylor(atan(x),   x, 0, 8))$
print("sqrt(1+x) =", taylor(sqrt(1+x), x, 0, 8))$
print("")$
print("Taylor γύρω από x0 = 1 για το e^x:")$
print("e^x =", taylor(exp(x), x, 1, 6))$
print("   (δυνάμεις του (x-1), με σταθερό παράγοντα %e)")$

print("")$
print("============================================================")$
print("Γ. ΠΡΑΞΕΙΣ: ΠΑΡΑΓΩΓΙΣΗ ΚΑΙ ΟΛΟΚΛΗΡΩΣΗ ΟΡΟ-ΟΡΟ")$
print("============================================================")$

/* Σ x^n = 1/(1-x) για |x| < 1 */
T   : taylor(1/(1-x), x, 0, 6)$
print("Σ x^n = 1/(1-x) :", T)$
dT  : expand(diff(T, x))$
ref1: expand(ratdisrep(taylor(1/(1-x)^2, x, 0, 5)))$
print("d/dx [Σ x^n] = Σ n x^(n-1) =", dT)$
print("ανάπτυγμα του 1/(1-x)^2    =", ref1)$
print("ταυτίζονται:", is(equal(dT, ref1)))$

print("")$
U   : taylor(1/(1+x), x, 0, 7)$
IU  : expand(integrate(ratdisrep(U), x))$
ref2: expand(ratdisrep(taylor(log(1+x), x, 0, 8)))$
print("Σ (-1)^n x^n = 1/(1+x) :", U)$
print("∫ [Σ (-1)^n x^n] dx = Σ (-1)^n x^(n+1)/(n+1) =", IU)$
print("ανάπτυγμα του log(1+x)                       =", ref2)$
print("ταυτίζονται:", is(equal(IU, ref2)))$

print("")$
print("============================================================")$
print("Δ. ΥΠΟΛΟΓΙΣΜΟΣ ΤΟΥ π")$
print("============================================================")$

/* Gregory-Leibniz: η σειρά του atan για x = 1,  π/4 = Σ (-1)^k/(2k+1) */
GL(N) := 4 * ev(sum((-1)^k/(2*k+1), k, 0, N), simpsum)$
print("Gregory-Leibniz  π/4 = Σ_{k=0}^N (-1)^k/(2k+1):")$
for N in [10, 100, 1000] do
  print("   N =", N, ": π ~", float(GL(N)),
        "   σφάλμα =", float(abs(GL(N) - %pi)))$
print("   Το σφάλμα φθίνει σαν 1/N: χρειάζονται ~10^k όροι για k ψηφία.")$

print("")$
M : 4*(4*atan(1/5) - atan(1/239))$
print("Τύπος Machin  π/4 = 4 atan(1/5) - atan(1/239):")$
print("   M =", float(M), "   π =", float(%pi))$
print("   ακριβής ταυτότητα M = %pi :", is(equal(M, %pi)))$
print("   Οι σειρές του atan(1/5) και atan(1/239) συγκλίνουν γεωμετρικά,")$
print("   με λόγους 1/25 και 1/57121 — γι' αυτό η μέθοδος είναι ταχύτατη.")$

print("")$
print("============================================================")$
print("Ε. ∫_0^1 sin(x^2) dx ΜΕΣΩ ΔΥΝΑΜΟΣΕΙΡΑΣ")$
print("============================================================")$

/* sin(x^2) = Σ (-1)^k x^(4k+2)/(2k+1)!  ->  ∫_0^1 = Σ (-1)^k/((4k+3)(2k+1)!) */
exact : float(quad_qags(sin(x^2), x, 0, 1)[1])$
print("Αριθμητική τιμή (quad_qags) =", exact, "   (0.3102683017233811)")$
print("")$
print("Σφάλμα ανά τάξη αναπτύγματος:")$
for m in [2, 6, 10, 14, 18] do
  block([Tm, approx],
    Tm : ratdisrep(taylor(sin(x^2), x, 0, m)),
    approx : float(integrate(Tm, x, 0, 1)),
    print("   τάξη", m, ": προσέγγιση =", approx,
          "   σφάλμα =", abs(approx - exact)))$
print("Ήδη με τάξη 14 (4 όροι) πετυχαίνουμε ~7 σωστά δεκαδικά ψηφία.")$

/* Γραφήματα (άκρα σε αύξουσα σειρά) */
p1 : ratdisrep(taylor(exp(x), x, 0, 1))$
p3 : ratdisrep(taylor(exp(x), x, 0, 3))$
p5 : ratdisrep(taylor(exp(x), x, 0, 5))$
plot2d([exp(x), p1, p3, p5], [x, -3, 3], [y, -1, 12],
       [legend, "e^x", "Taxi 1", "Taxi 3", "Taxi 5"],
       [xlabel, "x"], [ylabel, "y"],
       [title, "Maclaurin tou e^x"])$

q3 : ratdisrep(taylor(log(1+x), x, 0, 3))$
q7 : ratdisrep(taylor(log(1+x), x, 0, 7))$
plot2d([log(1+x), q1, q3, q7], [x, -0.9, 1.5], [y, -3, 2],
       [legend, "log(1+x)", "Taxi 1", "Taxi 3", "Taxi 7"],
       [xlabel, "x"], [ylabel, "y"],
       [title, "R = 1: ektos tou (-1,1) ta polyonyma apoklinoun"])$
