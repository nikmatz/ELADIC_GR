/* ── Μέρος Α: Εξέλιξη x_{k+1} = A x_k ─────────────────────── */
print("════ Α. Εξέλιξη Δυναμικού Συστήματος ════");
A: matrix([0.8, 0], [0, 0.6]);
eigenvalues(A);
x0: matrix([1],[1]);
x1: A . x0;
x2: A . x1;
x10: float(A^^10 . x0);
/* ρ(A) < 1 → x_k → 0 : ασυμπτωτικά σταθερό */

/* ── Μέρος Β: Σταθερότητα μέσω ιδιοτιμών ──────────────────── */
print("════ Β. Ιδιοτιμές και Σταθερότητα ════");
B: matrix([1.1, 0.2], [0.1, 0.9]);
ev: eigenvalues(B);
float(ev);
/* |λ|max > 1 → ασταθές: οι τροχιές απομακρύνονται */

/* ── Μέρος Γ: Αλυσίδα Markov — σταθερή κατανομή ───────────── */
print("════ Γ. Αλυσίδα Markov ════");
P: matrix([0.9, 0.2], [0.1, 0.8]);
[vals, vecs]: eigenvectors(P);
/* ιδιοδιάνυσμα για λ=1, κανονικοποίηση σε άθροισμα 1 */
v: transpose(matrix(vecs[1][1]));
pi_st: v / (v[1,1] + v[2,1]);
float(pi_st);

/* ── Μέρος Δ: A^k μέσω διαγωνοποίησης ─────────────────────── */
print("════ Δ. Κλειστός τύπος A^k ════");
[valsA, vecsA]: eigenvectors(A);
Pm: transpose(matrix(vecsA[1][1], vecsA[2][1]));
D:  matrix([valsA[1][1], 0], [0, valsA[1][2]]);
Ak: Pm . D^^k . invert(Pm);
ratsimp(Ak);
