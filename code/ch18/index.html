Επιλέξτε γλώσσα από τις καρτέλες. Τον κώδικα Python μπορείτε να τον τρέξετε εδώ μέσα στη σελίδα (κουμπί «▶ Τρέξε») — μπορείτε και να τον επεξεργαστείτε. Ο κώδικας Maxima εκτελείται στο wxMaxima.

Βασικές εντολές: integrate(f,x,a,inf)   gamma(n)   beta(m,n)   romberg(f,x,a,b) 
/* maxima_genikeum.mac — Γενικευμένα Ολοκληρώματα (Κεφ. 18) */
/* Ματζάκος, Ν. (2026). Στοιχεία Γραμμικής Άλγεβρας, Διαφορικού & Ολοκληρωτικού Λογισμού. NewTech Publications. */
/* Εντολές: integrate(f,x,a,inf), limit(F,t,inf), gamma(n), beta(m,n), romberg(f,x,a,b) */

/* Το γενικευμένο ολοκλήρωμα ορίζεται ως ΟΡΙΟ ενός ορισμένου ολοκληρώματος.
   Αν το υπολογίσουμε έτσι, η Maxima επιστρέφει inf/und αντί να σταματήσει
   με σφάλμα "integral is divergent", και μπορούμε να το ερμηνεύσουμε.      */

assume(tt > 1)$
assume(ee > 0, ee < 1)$

/* A' είδους: ∫_1^inf f dx = lim_{t->inf} ∫_1^t f dx */
Iinf(f) := limit(integrate(f, x, 1, tt), tt, inf)$
/* B' είδους: ∫_0^1 f dx = lim_{e->0+} ∫_e^1 f dx (ιδιαίτερο σημείο στο 0) */
Izero(f) := limit(integrate(f, x, ee, 1), ee, 0, plus)$

print("============================================================")$
print("Α. ΣΥΓΚΛΙΣΗ / ΑΠΟΚΛΙΣΗ — το κριτήριο του p")$
print("============================================================")$

/* ∫_1^inf dx/x^p : συγκλίνει μόνο για p > 1 */
print("∫_1^inf dx/x^p :")$
for p in [1/2, 1, 2] do
  print("   p =", p, " ->", Iinf(x^(-p)))$
/* αναμενόμενα: p=1/2 -> inf, p=1 -> inf, p=2 -> 1 */

/* ∫_0^1 dx/sqrt(x) : ιδιαίτερο σημείο στο x=0, συγκλίνει (=2) */
print("")$
print("∫_0^1 dx/sqrt(x) =", Izero(1/sqrt(x)), "   (αναμενόμενο 2)")$

/* Ερμηνεία των συμβόλων που επιστρέφει η Maxima */
print("")$
print("ΕΡΜΗΝΕΙΑ ΣΥΜΒΟΛΩΝ:")$
print("   inf  = +οο : το ολοκλήρωμα απειρίζεται (απόκλιση)")$
print("          π.χ. limit(log(tt),tt,inf) =", limit(log(tt), tt, inf))$
print("   ind  = φραγμένο αλλά χωρίς όριο (ταλάντωση)")$
print("          π.χ. limit(sin(tt),tt,inf) =", limit(sin(tt), tt, inf))$
print("   und  = μη ορισμένο (μη φραγμένη ταλάντωση)")$
print("          π.χ. limit(tt*sin(tt),tt,inf) =", limit(tt*sin(tt), tt, inf))$
print("   Μόνο ένας ΠΕΠΕΡΑΣΜΕΝΟΣ αριθμός σημαίνει σύγκλιση.")$

print("")$
print("============================================================")$
print("Β. ΣΥΝΑΡΤΗΣΕΙΣ ΓΑΜΜΑ ΚΑΙ ΒΗΤΑ")$
print("============================================================")$

print("Γ(1/2) =", gamma(1/2), "=", float(gamma(1/2)),
      "   sqrt(%pi) =", float(sqrt(%pi)))$
print("   ισότητα Γ(1/2) = sqrt(%pi) :", is(equal(gamma(1/2), sqrt(%pi))))$
print("Γ(5/2) =", gamma(5/2), "=", float(gamma(5/2)),
      "   (3*sqrt(%pi)/4 = 1.32934038818)")$
print("Γ(5)   =", gamma(5), " = 4! =", 4!, "   (24)")$
print("B(2,3) =", beta(2,3), "=", float(beta(2,3)), "   (1/12)")$
print("Γ(2)Γ(3)/Γ(5) =", gamma(2)*gamma(3)/gamma(5))$
print("   ισότητα B(m,n) = Γ(m)Γ(n)/Γ(m+n) :",
      is(equal(beta(2,3), gamma(2)*gamma(3)/gamma(5))))$

/* Η Γάμμα ως γενικευμένο ολοκλήρωμα: Γ(s) = ∫_0^inf x^(s-1) e^(-x) dx */
print("")$
print("∫_0^inf x^(1/2-1) e^(-x) dx =",
      integrate(x^(1/2-1)*exp(-x), x, 0, inf), "   (= Γ(1/2) = sqrt(%pi))")$

print("")$
print("============================================================")$
print("Γ. ΤΟ ΟΛΟΚΛΗΡΩΜΑ GAUSS — ΣΥΜΒΟΛΙΚΑ ΚΑΙ ΑΡΙΘΜΗΤΙΚΑ")$
print("============================================================")$

load(romberg)$

G : integrate(exp(-x^2), x, 0, inf)$
print("∫_0^inf e^(-x^2) dx =", G, "=", float(G))$
print("   sqrt(%pi)/2      =", float(sqrt(%pi)/2))$
print("   ισότητα :", is(equal(G, sqrt(%pi)/2)))$

/* Αριθμητικά σε πεπερασμένο διάστημα (romberg = η "quad" της Maxima) */
print("")$
print("∫_0^5 e^(-x^2) dx (romberg) =", romberg(exp(-x^2), x, 0, 5))$
print("   (η ουρά ∫_5^inf e^(-x^2) dx είναι ~ 1.4e-12, άρα πρακτικά sqrt(%pi)/2)")$
print("   |romberg - sqrt(%pi)/2| =",
      abs(romberg(exp(-x^2), x, 0, 5) - float(sqrt(%pi)/2)),
      " (η διαφορά είναι η ανοχή της romberg, όχι η ουρά)")$

print("")$
print("============================================================")$
print("Δ. ΚΡΙΤΗΡΙΟ ΣΥΓΚΡΙΣΗΣ:  0 < e^(-x)/x <= e^(-x)  για x >= 1")$
print("============================================================")$

Emin : Iinf(exp(-x))$
print("∫_1^inf e^(-x) dx     =", Emin, "=", float(Emin),
      "   (e^(-1) = 0.36787944117144233)")$
print("∫_1^20 e^(-x)/x dx    =", romberg(exp(-x)/x, x, 1, 20),
      "   (E1(1) = 0.2193839344)")$
print("")$
print("Αφού 0 < e^(-x)/x <= e^(-x) για κάθε x >= 1 και το μεγαλύτερο")$
print("ολοκλήρωμα συγκλίνει (= 1/e), συγκλίνει και το μικρότερο.")$
print("Αριθμητικός έλεγχος του φραγμού σε λίγα σημεία:")$
for xv in [1, 2, 5, 10] do
  print("   x =", xv, ": e^(-x)/x =", float(exp(-xv)/xv),
        " <= e^(-x) =", float(exp(-xv)))$

/* Γράφημα σύγκρισης (άκρα σε αύξουσα σειρά) */
plot2d([1/x^2, 1/x, 1/sqrt(x)], [x, 1, 10], [y, 0, 1.2],
       [legend, "1/x^2 (συγκλίνει)", "1/x (αποκλίνει)", "1/sqrt(x) (αποκλίνει)"],
       [xlabel, "x"], [ylabel, "f(x)"],
       [title, "Kritirio p sto [1,inf)"])$

plot2d([exp(-x)/x, exp(-x)], [x, 1, 8], [y, 0, 0.4],
       [legend, "e^(-x)/x", "e^(-x)"],
       [xlabel, "x"], [ylabel, "f(x)"],
       [title, "Kritirio sygkrisis"])$
